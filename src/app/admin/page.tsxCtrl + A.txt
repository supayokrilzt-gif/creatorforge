"use client";

import { useEffect, useState } from "react";
import type { User } from "@supabase/supabase-js";
import { supabase } from "@/lib/supabase";

type Member = {
  id: string;
  email: string;
  display_name: string | null;
  credits: number;
  created_at: string;
};

export default function AdminPage() {
  const [user, setUser] = useState<User | null>(null);
  const [checking, setChecking] = useState(true);
  const [isAdmin, setIsAdmin] = useState(false);

  const [email, setEmail] = useState("");
  const [member, setMember] = useState<Member | null>(null);

  const [amount, setAmount] = useState("10");
  const [note, setNote] = useState("");

  const [message, setMessage] = useState("");
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    checkAdmin();
  }, []);

  async function checkAdmin() {
    setChecking(true);

    const {
      data: { session },
    } = await supabase.auth.getSession();

    const currentUser = session?.user ?? null;

    setUser(currentUser);

    if (!currentUser) {
      setIsAdmin(false);
      setChecking(false);
      return;
    }

    const { data, error } = await supabase.rpc(
      "is_current_user_admin"
    );

    if (error) {
      console.error(error);
      setIsAdmin(false);
      setChecking(false);
      return;
    }

    setIsAdmin(Boolean(data));
    setChecking(false);
  }

  async function findMember() {
    setMessage("");
    setMember(null);

    const cleanEmail = email.trim();

    if (!cleanEmail) {
      setMessage("กรุณากรอกอีเมลสมาชิก");
      return;
    }

    setLoading(true);

    const { data, error } = await supabase.rpc(
      "admin_find_member",
      {
        p_email: cleanEmail,
      }
    );

    setLoading(false);

    if (error) {
      console.error(error);
      setMessage("ค้นหาสมาชิกไม่สำเร็จ");
      return;
    }

    if (!data || data.length === 0) {
      setMessage("ไม่พบสมาชิกอีเมลนี้");
      return;
    }

    const found = data[0];

    setMember({
      id: found.id,
      email: found.email,
      display_name: found.display_name,
      credits: found.credits,
      created_at: found.created_at,
    });
  }

  async function changeCredits(value: number) {
    if (!member) {
      setMessage("กรุณาค้นหาสมาชิกก่อน");
      return;
    }

    if (value === 0) {
      setMessage("จำนวนเครดิตต้องไม่เป็น 0");
      return;
    }

    setLoading(true);
    setMessage("");

    const { data, error } = await supabase.rpc(
      "admin_change_credits",
      {
        p_user_id: member.id,
        p_amount: value,
        p_note:
          note.trim() ||
          (value > 0
            ? "เติมเครดิตโดยแอดมิน"
            : "หักเครดิตโดยแอดมิน"),
      }
    );

    setLoading(false);

    if (error) {
      console.error(error);

      if (
        error.message?.includes(
          "INVALID_BALANCE"
        )
      ) {
        setMessage(
          "หักไม่ได้ เพราะเครดิตจะติดลบ"
        );
        return;
      }

      setMessage(
        "เปลี่ยนเครดิตไม่สำเร็จ"
      );
      return;
    }

    const newBalance = Number(data);

    setMember({
      ...member,
      credits: newBalance,
    });

    setMessage(
      value > 0
        ? `เติมเครดิตสำเร็จ +${value} เครดิต`
        : `หักเครดิตสำเร็จ ${value} เครดิต`
    );
  }

  async function applyCustomAmount(
    mode: "add" | "remove"
  ) {
    const parsed = Number(amount);

    if (
      !Number.isFinite(parsed) ||
      parsed <= 0
    ) {
      setMessage(
        "กรุณากรอกจำนวนเครดิตมากกว่า 0"
      );
      return;
    }

    const value =
      mode === "add"
        ? Math.floor(parsed)
        : -Math.floor(parsed);

    await changeCredits(value);
  }

  async function logout() {
    await supabase.auth.signOut();
    window.location.href = "/";
  }

  if (checking) {
    return (
      <main style={styles.page}>
        <div style={styles.centerCard}>
          <h2 style={styles.gold}>
            กำลังตรวจสอบสิทธิ์...
          </h2>
        </div>
      </main>
    );
  }

  if (!user) {
    return (
      <main style={styles.page}>
        <div style={styles.centerCard}>
          <div style={styles.icon}>
            🔒
          </div>

          <h1 style={styles.gold}>
            Admin Login Required
          </h1>

          <p style={styles.muted}>
            กรุณาเข้าสู่ระบบจากหน้า CreatorForge ก่อน
          </p>

          <button
            style={styles.mainButton}
            onClick={() =>
              (window.location.href = "/")
            }
          >
            กลับหน้าหลัก
          </button>
        </div>
      </main>
    );
  }

  if (!isAdmin) {
    return (
      <main style={styles.page}>
        <div style={styles.centerCard}>
          <div style={styles.icon}>
            ⛔
          </div>

          <h1 style={styles.gold}>
            ไม่มีสิทธิ์เข้าถึง
          </h1>

          <p style={styles.muted}>
            บัญชีนี้ไม่ใช่ผู้ดูแลระบบ
          </p>

          <button
            style={styles.mainButton}
            onClick={() =>
              (window.location.href = "/")
            }
          >
            กลับหน้าหลัก
          </button>
        </div>
      </main>
    );
  }

  return (
    <main style={styles.page}>
      <header style={styles.header}>
        <div>
          <div style={styles.logo}>
            ✦ CREATORFORGE
          </div>

          <div style={styles.adminTitle}>
            ADMIN PANEL
          </div>
        </div>

        <div style={styles.headerRight}>
          <span style={styles.adminBadge}>
            ADMIN
          </span>

          <span style={styles.email}>
            {user.email}
          </span>

          <button
            style={styles.secondaryButton}
            onClick={() =>
              (window.location.href = "/")
            }
          >
            กลับหน้าเว็บ
          </button>

          <button
            style={styles.secondaryButton}
            onClick={logout}
          >
            ออกจากระบบ
          </button>
        </div>
      </header>

      <section style={styles.container}>
        <div style={styles.hero}>
          <div>
            <p style={styles.eyebrow}>
              CREATORFORGE CONTROL CENTER
            </p>

            <h1 style={styles.heroTitle}>
              จัดการเครดิตสมาชิก
            </h1>

            <p style={styles.muted}>
              ค้นหาสมาชิกด้วยอีเมล แล้วเติมหรือหักเครดิตได้ทันที
            </p>
          </div>
        </div>

        <section style={styles.card}>
          <h2 style={styles.sectionTitle}>
            ค้นหาสมาชิก
          </h2>

          <div style={styles.searchRow}>
            <input
              style={styles.input}
              type="email"
              placeholder="email@example.com"
              value={email}
              onChange={(e) =>
                setEmail(e.target.value)
              }
              onKeyDown={(e) => {
                if (e.key === "Enter") {
                  findMember();
                }
              }}
            />

            <button
              style={styles.mainButton}
              onClick={findMember}
              disabled={loading}
            >
              {loading
                ? "กำลังค้นหา..."
                : "ค้นหา"}
            </button>
          </div>

          {message && (
            <div style={styles.message}>
              {message}
            </div>
          )}
        </section>

        {member && (
          <section style={styles.card}>
            <div style={styles.memberTop}>
              <div>
                <p style={styles.eyebrow}>
                  MEMBER
                </p>

                <h2 style={styles.memberName}>
                  {member.display_name ||
                    "สมาชิก"}
                </h2>

                <p style={styles.memberEmail}>
                  {member.email}
                </p>
              </div>

              <div style={styles.creditBox}>
                <span style={styles.creditLabel}>
                  เครดิตปัจจุบัน
                </span>

                <strong
                  style={styles.creditNumber}
                >
                  {member.credits}
                </strong>
              </div>
            </div>

            <div style={styles.quickSection}>
              <h3 style={styles.subTitle}>
                เติมเครดิตด่วน
              </h3>

              <div style={styles.quickButtons}>
                {[10, 30, 50, 100].map(
                  (value) => (
                    <button
                      key={value}
                      style={styles.quickButton}
                      onClick={() =>
                        changeCredits(value)
                      }
                      disabled={loading}
                    >
                      +{value}
                    </button>
                  )
                )}
              </div>
            </div>

            <div style={styles.divider} />

            <div style={styles.customGrid}>
              <div>
                <label style={styles.label}>
                  จำนวนเครดิต
                </label>

                <input
                  style={styles.input}
                  type="number"
                  min="1"
                  value={amount}
                  onChange={(e) =>
                    setAmount(
                      e.target.value
                    )
                  }
                />
              </div>

              <div>
                <label style={styles.label}>
                  หมายเหตุ
                </label>

                <input
                  style={styles.input}
                  placeholder="เช่น ชำระเงินแล้ว"
                  value={note}
                  onChange={(e) =>
                    setNote(
                      e.target.value
                    )
                  }
                />
              </div>
            </div>

            <div style={styles.actionRow}>
              <button
                style={styles.addButton}
                onClick={() =>
                  applyCustomAmount("add")
                }
                disabled={loading}
              >
                + เติมเครดิต
              </button>

              <button
                style={styles.removeButton}
                onClick={() =>
                  applyCustomAmount(
                    "remove"
                  )
                }
                disabled={loading}
              >
                − หักเครดิต
              </button>
            </div>

            <p style={styles.memberDate}>
              สมัครสมาชิก:{" "}
              {new Date(
                member.created_at
              ).toLocaleString("th-TH")}
            </p>
          </section>
        )}
      </section>
    </main>
  );
}
